;; Dated measurement data; append to generator ca2955ee... for explicit local observation.
;; Never imported by native builds, gates or CI.


(def oracle-out (.-KOTOBA_NATIVE_ORACLE_OUT js/process.env))
(fs/mkdirSync oracle-out #js {:recursive true})
(defn oracle-write [name node] (fs/writeFileSync (path/join oracle-out name) (html/->html node)))
(def identity-view (org-identity-view))
(def identity-article (first (filter #(and (vector? %) (= :article (first %))) (tree-seq coll? seq identity-view))))
(when-not identity-article (throw (js/Error. "missing identity article")))
(oracle-write "identity-article.html" identity-article)
(fs/writeFileSync (path/join oracle-out "identity-metadata.json") (js/JSON.stringify (clj->js (select-keys org-identity-copy [:title :description]))))
(doseq [tag ["en" "ja"]
        article-path ["/blog/embench-c-parity-20261015/" "/blog/embench-progress-20261012/" "/blog/kotoba-murakumo-how-to-use-20261015/"]]
  (let [catalog (js->clj (js/JSON.parse (fs/readFileSync (str "site/i18n/" tag ".json") "utf8")))
        tr (fn [s] (let [v (or (get catalog (message-id s)) (throw (js/Error. (str "missing chrome message " s))))] (str (or (re-find #"^\s+" s) "") v (or (re-find #"\s+$" s) ""))))
        translated-paths (set (map :path localized-pages))
        resolve-link (fn [href]
                       (if (or (str/starts-with? href "#") (re-find #"^(?:[a-z]+:|//)" href)) href
                         (let [u (js/URL. href (str site-origin article-path)) p (.-pathname u)]
                           (str (if (translated-paths p) (str "/" tag p) p) (.-search u) (.-hash u)))))
        localize (fn [node] (locales/mark-language (if (= tag "en") node (-> node (locales/map-text tr) (locales/rebase-links resolve-link))) tag))
        slug (second (re-find #"/blog/([^/]+)/" article-path))]
    (oracle-write (str tag "-" slug "-header.html") (localize (header "/" article-path #{"en" "ja"})))
    (oracle-write (str tag "-" slug "-footer.html") (localize (footer :en "/legal/")))))
(println "Recorded explicit identity-body and localized-chrome source observations")
